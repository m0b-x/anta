# A delete meets an edit it has not seen. With both guests in sync, the
# pass's guest writes "Meadow hold" into Session 2 and syncs it; the second
# guest (still manual), which never pulled that version, deletes its
# Session 2 and drains: the push is held, the stored version merged
# first — its text filed as a conflict copy — and the delete goes again
# (`drain --expect held`). On both guests, after a sync each, Session 2 is
# gone and the copy holds the pass's guest's line. The second guest goes
# back to the automatic cadence at the end.
# - Session 2 is found by "Harbor plank", which 01 wrote into it and the
#   copy carries too: one hit for it after the delete — the copy — is
#   Session 2 gone, so the tap on the hit's path is the check.
# - Deleted from Week 1, as share/04_o deletes it: Session 2 sorts first
#   there (last updated, by 01's edit), and the confirm names it.
# Both in sync first.
-d emulator-5558 tap id:sharing-sync-now
-d emulator-5558 expect id:sharing-sync-status "Syncs when you tap" --absent "waiting" --timeout 30
tap id:sharing-sync-now --settle 3000
expect id:sharing-sync-status "Up to date" --timeout 60
# The pass's guest: its line into Session 2, then "Sync now".
key back
wait id:create-folder
tap id:search-open
wait id:search-field
tap id:search-field
type "Harbor plank"
expect --absent "Recent" --timeout 10
wait "Training › Week 1" --timeout 10
tap "Training › Week 1"
wait id:editor-toolbar --timeout 15
tap id:editor-body
