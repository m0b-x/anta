# The owner shares one event of two. After a sync on each side the
# partner's calendar lists it among what is coming up — and not the event
# that was not shared. Its category is one the partner's database does not
# have, so it draws under the fallback.
share event "Lifting session"
tap id:sharing-sync-now --settle 3000
expect id:sharing-sync-status "Up to date" --timeout 60
-d emulator-5558 tap id:sharing-sync-now
-d emulator-5558 expect id:sharing-sync-status "Up to date" --timeout 60
# The partner's calendar, from the root.
-d emulator-5558 key back
-d emulator-5558 wait id:settings-bar-hub
-d emulator-5558 key back
-d emulator-5558 wait id:create-folder
-d emulator-5558 tap id:nav-menu
-d emulator-5558 wait id:drawer-calendar
-d emulator-5558 tap id:drawer-calendar
-d emulator-5558 wait id:calendar-add-event
-d emulator-5558 tap id:panel-mode-day --settle 1000
-d emulator-5558 tap id:panel-mode-upcoming
-d emulator-5558 expect "Lifting session" --absent "Yearly checkup" --timeout 30
-d emulator-5558 shot shared-event
# Back to Sync & sharing, where the next flow starts.
-d emulator-5558 key back
-d emulator-5558 wait id:create-folder
-d emulator-5558 tap id:nav-menu
-d emulator-5558 wait id:drawer-account
-d emulator-5558 tap id:drawer-account
-d emulator-5558 wait id:settings-bar-sharing
-d emulator-5558 errors
errors
