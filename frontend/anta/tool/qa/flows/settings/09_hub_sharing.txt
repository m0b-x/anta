# Sharing through the hub, left by the bar's arrow and then by system back,
# each landing on the hub and then on the folder root (D2, D3). The QA build
# runs without cloud (SyncAvailability.resolve is false for a QA build that
# does not allow it), so the page reads SyncUnavailable and opens on Not
# available on this platform: on the first visit its list shows that
# disabled row under its own id (S7), and the Sync section its disabled
# "Sync this database" row. A QA build with cloud allowed would open signed
# out, on sharing-sign-in; the `sync` suite covers that build.
tap id:nav-menu
wait id:drawer-calendar
scroll-to id:drawer-settings --in id:drawer-list
tap id:drawer-settings
wait id:settings-bar-hub
tap id:settings-hub-row-syncSettings
wait id:settings-bar-sharing
expect id:sharing-list id:sharing-unavailable id:sharing-sync
tap id:nav-back
wait id:settings-bar-hub
expect --absent id:settings-bar-sharing
tap id:nav-back
wait id:create-folder
expect --absent id:settings-bar-hub --absent id:drawer-settings
tap id:nav-menu
wait id:drawer-calendar
scroll-to id:drawer-settings --in id:drawer-list
tap id:drawer-settings
wait id:settings-bar-hub
tap id:settings-hub-row-syncSettings
wait id:settings-bar-sharing
key back
wait id:settings-bar-hub
expect --absent id:settings-bar-sharing
key back

expect --absent id:settings-bar-hub --absent id:drawer-settings
