# Restore after a relaunch for the four stamped pages whose arrow raises the
# drawer today — App settings, Databases, Sharing, Counters: each opened
# from the drawer, the app relaunched on the same database (a bare
# `relaunch`), the page back on screen, and system back down to the folder
# root with the drawer closed (a restored page's continuation raises the
# drawer only on the arrow's openDrawer result, never on system back).
# - The stack is written 400 ms after the push (NavigationHistoryService's
#   debounce) and a relaunch force-stops the app with no pause to flush on,
#   so the row's tap holds 2 s before the relaunch can run.
# - A relaunch is a cold start: the restored page is waited for 20 s.
tap id:nav-menu
wait id:drawer-calendar
tap id:drawer-settings --settle 2000
wait id:settings-bar-app
relaunch
wait id:settings-bar-app --timeout 20
key back
wait id:create-folder
expect --absent id:settings-bar-app --absent id:drawer-settings
tap id:nav-menu
wait id:drawer-calendar
scroll-to id:settings-databases
tap id:settings-databases --settle 2000
wait id:settings-bar-databases
relaunch
wait id:settings-bar-databases --timeout 20
key back
wait id:create-folder
expect --absent id:settings-bar-databases --absent id:drawer-settings
tap id:nav-menu
wait id:drawer-calendar
tap id:drawer-sharing --settle 2000
wait id:settings-bar-sharing
relaunch
wait id:settings-bar-sharing --timeout 20
key back
wait id:create-folder
expect --absent id:settings-bar-sharing --absent id:drawer-settings
tap id:nav-menu
wait id:drawer-calendar
tap id:drawer-counters --settle 2000
wait id:settings-bar-counters
relaunch
wait id:settings-bar-counters --timeout 20
key back
wait id:create-folder
expect --absent id:settings-bar-counters --absent id:drawer-settings
errors
