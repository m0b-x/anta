# Restore after a relaunch for the four calendar pages the hub opens —
# Calendar settings, Appearance, Categories, the guide: each opened from the
# hub, the app relaunched on the same database (a bare `relaunch`), the page
# back on screen over the hub ([settingsHub, page], D3), and system back
# down through the hub to the folder root with the drawer closed.
# - The stack is written 400 ms after the push (NavigationHistoryService's
#   debounce) and a relaunch force-stops the app with no pause to flush on,
#   so the row's tap holds 2 s before the relaunch can run.
# - A relaunch is a cold start: the restored page is waited for 20 s.
tap id:nav-menu
wait id:drawer-calendar
scroll-to id:drawer-settings --in id:drawer-list
tap id:drawer-settings
wait id:settings-bar-hub
tap id:settings-hub-row-calendarSettings --settle 2000
wait id:settings-bar-calendar
relaunch
wait id:settings-bar-calendar --timeout 20
key back
wait id:settings-bar-hub
key back
wait id:create-folder
expect --absent id:settings-bar-hub --absent id:drawer-settings
tap id:nav-menu
wait id:drawer-calendar
scroll-to id:drawer-settings --in id:drawer-list
tap id:drawer-settings
wait id:settings-bar-hub

wait id:appearance-today-style
relaunch
wait id:appearance-today-style --timeout 20
key back
wait id:settings-bar-hub
key back
wait id:create-folder
expect --absent id:settings-bar-hub --absent id:drawer-settings
tap id:nav-menu
wait id:drawer-calendar
scroll-to id:drawer-settings --in id:drawer-list
tap id:drawer-settings
wait id:settings-bar-hub
tap id:settings-hub-row-calendarCategories --settle 2000
