# The drawer from the folder root lists places (D1): the nine rows by id,
# the settings rows that moved to the hub absent, then closed by system back.
# - The rows live in a ListView, so the lower ones may sit past the cache
#   extent and out of the tree: the second expect runs after a scroll-to,
#   swiped inside the list (--in id:drawer-list) — the drawer's first
#   scrollable in a dump is the header's avatar block, which scrolls nothing.
# - The moved rows are read absent by their English labels after the scroll,
#   with the whole list in the tree, so an absence means gone rather than far
#   away; "Settings" is never one of them (the header's subtitle reads it).
wait id:create-folder
tap id:nav-menu
wait id:drawer-calendar
expect id:drawer-calendar-overview id:drawer-calendar-report id:drawer-calendar-categories id:drawer-alerts id:drawer-counters
look 00_drawer
scroll-to id:drawer-about --in id:drawer-list
expect id:drawer-settings id:drawer-how-to id:drawer-about --absent "Calendar settings" --absent "Markdown shortcuts" --absent "Sharing" --absent "Databases" --absent "Language" --absent id:settings-appearance --absent "Developer options"
key back
wait id:create-folder
expect --absent id:drawer-settings
errors
