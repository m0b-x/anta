# "Delete cloud copy" (cloud-multi-user roadmap D1): its dialog's own
# confirm, then the Sync section unbound at once and the deletion's row in
# a section of its own, last on the page. On a database this small the
# background run ends within a second, so the row is waited for by its id,
# never by "Deleting cloud copy": it ends as "Cloud copy deleted" and stays,
# no status row left. Leaving the page and opening it again shows no
# deletion row at all. The tap waits for the row, so the later checks are
# never vacuous.
expect id:sharing-sync-status "Up to date" --timeout 60
tap id:sharing-sync-delete
wait id:sharing-sync-delete-confirm
tap id:sharing-sync-delete-confirm --wait id:sharing-sync-finish-delete --wait-timeout 5
expect id:sharing-sync --absent id:sharing-sync-status --absent id:sharing-sync-delete
expect id:sharing-sync-finish-delete "Cloud copy deleted" --absent "Finish deleting" --timeout 60
expect id:sharing-sync id:sharing-sync-finish-delete --absent id:sharing-sync-status --absent id:sharing-sync-stop
shot deleted
key back
wait id:settings-bar-hub
tap id:settings-hub-row-syncSettings
wait id:settings-bar-sharing
expect id:sharing-sign-out id:sharing-sync --absent id:sharing-sync-finish-delete --absent id:sharing-sync-status
errors
